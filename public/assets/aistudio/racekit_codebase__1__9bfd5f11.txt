_sync_task: Optional[asyncio.Task] = None


async def _periodic_sync():
    """Background task: sync with Garmin every SYNC_INTERVAL_MINUTES."""
    while True:
        await asyncio.sleep(config.SYNC_INTERVAL_MINUTES * 60)
        try:
            log.info("Background sync starting...")
            syncer = get_syncer()
            await asyncio.to_thread(syncer.full_sync)
            log.info("Background sync complete")
        except Exception as e:
            log.error(f"Background sync failed: {e}")


# ── App lifecycle ────────────────────────────────────────────────

@asynccontextmanager
async def lifespan(app: FastAPI):
    # Startup
    log.info("🏃 RaceKit starting up...")
    db.init_db()

    # Seed from local cache if database has no records yet
    try:
        if len(db.get_best_efforts()) == 0:
            seed_from_cache()
    except Exception as e:
        log.warning(f"Seed note: {e}")

    # Initial sync (in background to not block startup)
    async def _initial_sync():
        try:
            syncer = get_syncer()
            connected = await asyncio.to_thread(syncer.connect)
            if connected:
                await asyncio.to_thread(syncer.full_sync)
                log.info("✅ Live Garmin sync complete")
            else:
                log.info("ℹ️ Running on baseline Garmin data")
        except Exception as e:
            log.warning(f"Initial sync: {e}")

    asyncio.create_task(_initial_sync())

    # Start periodic sync
    global _sync_task
    _sync_task = asyncio.create_task(_periodic_sync())

    yield

    # Shutdown
    if _sync_task:
        _sync_task.cancel()
    log.info("RaceKit shutting down")


app = FastAPI(title="RaceKit", version="1.0.0", lifespan=lifespan)

def _build_engine() -> PredictionEngine:
    """Build prediction engine from database efforts and training volume."""
    effort_rows = db.get_best_efforts()
    if not effort_rows:
        try:
            seed_from_cache()
            effort_rows = db.get_best_efforts()
        except Exception as e:
            log.warning(f"Auto-seed during engine build: {e}")

    health = db.get_latest_health()
    vol_stats = db.get_training_volume_stats()

    efforts = []
    for row in effort_rows:
        efforts.append(Effort(
            distance_m=row["distance_m"],
            time_s=row["time_s"],
            date=row["date"] or "",
            power_w=row.get("power_w"),
            hr_avg=row.get("hr_avg"),
            source=row.get("source", ""),
            label=row.get("label", ""),
        ))

    vo2max = health.get("vo2max") if health else None
    volume = VolumeProfile(
        longest_run_miles=vol_stats.get("longest_run_miles", 8.91),
        avg_weekly_miles=vol_stats.get("avg_weekly_miles", 17.1),
    )
    return PredictionEngine(efforts, vo2max=vo2max, volume=volume)


def _parse_conditions(
    elevation_gain=0.0,
    elevation_loss=None,
    surface: str = "road",
    altitude=0.0,
    temp_f=50.0,
    humidity=50.0,
    wind_mph=0.0,
    wind_dir: str = "head",
    shoe: str = "vaporfly_next2",
    hrv: str = "balanced",
    sleep: Optional[int] = None,
    battery: Optional[int] = None,
) -> CourseConditions:
    def _to_float(v, default=0.0):
        try:
            return float(v)
        except Exception:
            return default

    eg = _to_float(elevation_gain)
    el = _to_float(elevation_loss) if elevation_loss is not None else None
    alt = _to_float(altitude)
    tf = _to_float(temp_f, 50.0)
    hum = _to_float(humidity, 50.0)
    wnd = _to_float(wind_mph, 0.0)

    return CourseConditions(
        elevation_gain_m=eg * 0.3048,  # UI sends feet -> convert to meters
        elevation_loss_m=el * 0.3048 if el is not None else None,
        surface=str(surface),
        altitude_m=alt * 0.3048,       # UI sends feet -> convert to meters
        temperature_f=tf,
        humidity_pct=hum,
        wind_mph=wnd,
        wind_direction=str(wind_dir),
        shoe_type=str(shoe),
        hrv_status=str(hrv),
        sleep_score=sleep,
        body_battery=battery,
    )

@app.get("/api/predict")
async def predict_race(
    distance: str = "5K",
    elevation_gain: float = 0.0,
    elevation_loss: Optional[float] = None,
    surface: str = "road",
    altitude: float = 0.0,
    temp_f: float = 50.0,
    humidity: float = 50.0,
    wind_mph: float = 0.0,
    wind_dir: str = "head",
    shoe: str = "vaporfly_next2",
    use_readiness: bool = True,
):
    """Predict race time for a single distance with conditions."""
    try:
        engine = _build_engine()
        health = db.get_latest_health() if use_readiness else None

        # Resolve distance
        dist_m = STANDARD_DISTANCES.get(distance)
        if dist_m is None:
            try:
                dist_m = float(distance)
            except ValueError:
                return JSONResponse({"error": f"Unknown distance: {distance}"}, status_code=400)

        # Base prediction
        pred = engine.predict(dist_m, label=distance)

        # Apply conditions
        conditions = _parse_conditions(
            elevation_gain=elevation_gain,
            elevation_loss=elevation_loss,
            surface=surface,
            altitude=altitude,
            temp_f=temp_f,
            humidity=humidity,
            wind_mph=wind_mph,
            wind_dir=wind_dir,
            shoe=shoe,
            hrv=health.get("hrv_status", "balanced") if health else "balanced",
            sleep=health.get("sleep_score") if health else None,
            battery=health.get("body_battery") if health else None,
        )

        adjusted = apply_conditions(pred, conditions)

        return {
            "prediction": adjusted.to_dict(),
            "base_prediction": pred.to_dict(),
            "conditions": conditions.to_dict(),
        }
    except Exception as e:
        log.error(f"Prediction error: {e}", exc_info=True)
        return JSONResponse({"error": str(e)}, status_code=500)

_sync_state = {
    "syncing": False,
    "status": "idle",
    "message": "",
    "error": None,
    "timestamp": None,
}


@app.post("/api/sync")
async def trigger_sync(background_tasks: BackgroundTasks):
    """Trigger a manual Garmin sync."""
    if _sync_state["syncing"]:
        return JSONResponse({
            "status": "in_progress",
            "message": "Garmin sync is already in progress..."
        }, status_code=409)

    def _do_sync():
        _sync_state["syncing"] = True
        _sync_state["status"] = "in_progress"
        _sync_state["error"] = None
        _sync_state["message"] = "Connecting to Garmin Connect..."
        try:
            syncer = get_syncer()
            if not syncer.is_connected:
                ok = syncer.connect()
                if not ok:
                    raise ConnectionError(syncer.last_error or "Failed to authenticate with Garmin Connect")

            _sync_state["message"] = "Syncing activities, PRs, and recovery metrics..."
            res = syncer.full_sync()

            errs = res.get("errors", [])
            if errs and not res.get("activities") and not res.get("prs") and not res.get("health"):
                raise RuntimeError("; ".join(errs))

            db.log_sync("manual_sync", "success", f"Activities: {res.get('activities')}, PRs: {res.get('prs')}")
            _sync_state["status"] = "success"
            _sync_state["message"] = f"Synced {res.get('activities', 0)} activities & {res.get('prs', 0)} PRs"
            _sync_state["timestamp"] = datetime.now().isoformat()
            log.info("Manual Garmin sync completed successfully")
        except Exception as e:
            err_msg = str(e)
            log.error(f"Manual sync failed: {err_msg}")
            _sync_state["status"] = "error"
            _sync_state["error"] = err_msg
            _sync_state["message"] = f"Sync failed: {err_msg}"
            db.log_sync("manual_sync", "error", err_msg)
        finally:
            _sync_state["syncing"] = False

    background_tasks.add_task(_do_sync)
    return {"status": "sync_started", "message": "Syncing with Garmin in background..."}

    strategy: str = "course_gap",

    elevation_gain: float = 0.0,
    elevation_loss: Optional[float] = None,

    wind_mph: float = 0.0,
    wind_dir: str = "head",
    shoe: str = "vaporfly_next2",
    use_readiness: bool = True,
):
    """Generate actionable race splits adjusted for course terrain & conditions."""
    try:
        engine = _build_engine()
        health = db.get_latest_health() if use_readiness else None
        dist_m = STANDARD_DISTANCES.get(distance)
        if dist_m is None:
            dist_m = float(distance)

        pred = engine.predict(dist_m, label=distance)
        conditions = _parse_conditions(
            elevation_gain=elevation_gain,
            elevation_loss=elevation_loss,
            surface=surface,
            altitude=altitude,
            temp_f=temp_f,
            humidity=humidity,
            wind_mph=wind_mph,
            wind_dir=wind_dir,
            shoe=shoe,
            hrv=health.get("hrv_status", "balanced") if health else "balanced",
            sleep=health.get("sleep_score") if health else None,
            battery=health.get("body_battery") if health else None,
        )
        adjusted = apply_conditions(pred, conditions)

        profile = None
        if course_id:
            course = db.get_course(course_id)
            if course:
                profile = course.get("profile")

        splits = engine.generate_splits(adjusted, strategy=strategy, course_profile_ft=profile)
        return {
            "distance": distance,
            "strategy": strategy,
            "total_time_fmt": adjusted.format_time(),
            "splits": splits,
        }
    except Exception as e:
        log.error(f"Splits error: {e}", exc_info=True)
        return JSONResponse({"error": str(e)}, status_code=500)

@app.get("/api/shoe-matrix")
async def get_shoe_matrix(
    distance: str = "5K",
    elevation_gain: float = 0.0,
    elevation_loss: Optional[float] = None,
    surface: str = "road",
    altitude: float = 0.0,
    temp_f: float = 50.0,
    humidity: float = 50.0,
    wind_mph: float = 0.0,
    wind_dir: str = "head",
    use_readiness: bool = True,
):
    """Compare all 5 rotation shoes side-by-side for the current surface & conditions."""
    try:
        engine = _build_engine()
        health = db.get_latest_health() if use_readiness else None
        dist_m = STANDARD_DISTANCES.get(distance) or float(distance)
        base_pred = engine.predict(dist_m, label=distance)

        rotation_keys = ["dragonfly_xc", "vaporfly_next2", "dragonfly_track", "streakfly", "nb_1080"]
        results = []

        for key in rotation_keys:
            info = SHOE_DATABASE[key]
            cond = _parse_conditions(
                elevation_gain=elevation_gain,
                elevation_loss=elevation_loss,
                surface=surface,
                altitude=altitude,
                temp_f=temp_f,
                humidity=humidity,
                wind_mph=wind_mph,
                wind_dir=wind_dir,
                shoe=key,
                hrv=health.get("hrv_status", "balanced") if health else "balanced",
                sleep=health.get("sleep_score") if health else None,
                battery=health.get("body_battery") if health else None,
            )
            adj = apply_conditions(base_pred, cond)
            shoe_pct = adj.adjustments_applied.get("shoes", 0.0)
            results.append({
                "shoe_id": key,
                "label": info["label"],
                "time_s": round(adj.time_s, 2),
                "time_fmt": adj.format_time(),
                "pace_per_mile": adj.format_time(adj.pace_per_mile_s),
                "shoe_pct": shoe_pct,
            })

        results.sort(key=lambda r: r["time_s"])
        best_time = results[0]["time_s"]
        for i, r in enumerate(results):
            r["delta_s"] = round(r["time_s"] - best_time, 1)
            if i == 0:
                r["tag"] = "OPTIMAL"
            elif r["shoe_pct"] >= 5.0:
                r["tag"] = "SURFACE HAZARD"
            elif r["shoe_pct"] > 0:
                r["tag"] = "TRAINING HEAVY"
            else:
                r["tag"] = "VIABLE"

        return {"surface": surface, "distance": distance, "shoes": results}
    except Exception as e:

# New /api/weather endpoint (Open-Meteo, free & keyless):
@app.get("/api/weather")
async def get_course_weather(lat: float = 39.6079, lon: float = -75.7369):
    """Fetch current live race-day weather for a course's coordinates via Open-Meteo."""
    def _fetch():
        url = (
            f"https://api.open-meteo.com/v1/forecast?"
            f"latitude={lat}&longitude={lon}"
            f"&current=temperature_2m,relative_humidity_2m,wind_speed_10m,weather_code"
            f"&temperature_unit=fahrenheit&wind_speed_unit=mph"
        )
        r = requests.get(url, timeout=6)
        r.raise_for_status()
        return r.json()

    try:
        data = await asyncio.to_thread(_fetch)
        cur = data.get("current", {})
        return {
            "temp_f": round(float(cur.get("temperature_2m", 50.0))),
            "humidity": round(float(cur.get("relative_humidity_2m", 50.0))),
            "wind_mph": round(float(cur.get("wind_speed_10m", 0.0))),
            "weather_code": cur.get("weather_code", 0),
        }
    except Exception as e:
        log.warning(f"Weather fetch failed: {e}")
        return JSONResponse({"error": f"Weather unavailable: {e}"}, status_code=502)

"""SQLite database layer for RaceKit."""
import json

from datetime import datetime
from pathlib import Path
from typing import Any

import config

DB_PATH = config.DB_PATH


def _ensure_dir():
    DB_PATH.parent.mkdir(parents=True, exist_ok=True)


def _get_conn() -> sqlite3.Connection:
    _ensure_dir()
    conn = sqlite3.connect(str(DB_PATH))
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA journal_mode=WAL")
    conn.execute("PRAGMA foreign_keys=ON")
    return conn


@contextmanager
def get_db():
    conn = _get_conn()
    try:
        yield conn
        conn.commit()
    finally:
        conn.close()


def init_db():

def get_best_efforts() -> list[dict]:
    """Return the fastest effort at each standard distance."""
    with get_db() as db:
        rows = db.execute("""
            SELECT e1.* FROM efforts e1
            INNER JOIN (
                SELECT distance_m, MIN(time_s) as best_time
                FROM efforts
                GROUP BY distance_m
            ) e2 ON e1.distance_m = e2.distance_m AND e1.time_s = e2.best_time
            ORDER BY e1.distance_m
        """).fetchall()
        return [dict(r) for r in rows]

def get_training_volume_stats() -> dict:
    """Calculate rolling 6-week average weekly mileage and longest run distance."""
    with get_db() as db:
        row_max = db.execute("""
            SELECT MAX(distance_m), MAX(date(start_time))
            FROM activities
            WHERE activity_type LIKE '%run%'
        """).fetchone()
        longest_m = float(row_max[0]) if row_max and row_max[0] else 14332.0
        latest_date = row_max[1] if row_max and row_max[1] else None

        if latest_date:
            row_vol = db.execute("""
                SELECT COALESCE(SUM(distance_m), 0) / 1609.344 AS total_mi
                FROM activities
                WHERE activity_type LIKE '%run%'
                  AND date(start_time) >= date(?, '-42 days')
            """, (latest_date,)).fetchone()
            avg_mpw = float(row_vol["total_mi"]) / 6.0 if row_vol else 20.0
        else:
            avg_mpw = 20.0

        return {
            "longest_run_m": longest_m,
            "longest_run_miles": round(longest_m / 1609.344, 2),
            "avg_weekly_miles": round(avg_mpw, 1),
        }

def get_courses() -> list[dict]:
    """Return all courses (presets and user-imported Garmin courses)."""
    with get_db() as db:
        rows = db.execute("SELECT * FROM courses ORDER BY is_preset DESC, id ASC").fetchall()
        result = []
        for r in rows:
            d = dict(r)
            try:
                d["profile"] = json.loads(d.get("profile_json") or "[]")
            except Exception:
                d["profile"] = [0, 5, 2, 8, 4, 10, 6, 8, 3, 0]
            try:

                d["route"] = []
            try:
                d["mile_markers"] = json.loads(d.get("mile_markers_json") or "[]")
            except Exception:
                d["mile_markers"] = []
            result.append(d)
        return result

def get_garmin_candidates(limit: int = 40) -> list[dict]:
    """Return past running activities suitable for importing as race courses."""
    with get_db() as db:
        rows = db.execute("""
            SELECT garmin_id, name, activity_type, start_time, distance_m, duration_s,
                   elevation_gain, elevation_loss, avg_speed_mps, raw_json
            FROM activities
            WHERE (activity_type LIKE '%run%' OR distance_m >= 3000)
            ORDER BY start_time DESC
            LIMIT ?
        """, (limit,)).fetchall()

    candidates = []
    for r in rows:
        d = dict(r)
        dist_m = d.get("distance_m") or 0.0
        dur_s = d.get("duration_s") or 0.0
        dist_mi = dist_m / 1609.344
        dist_km = dist_m / 1000.0

        # Gain
        gain_m = d.get("elevation_gain") or 0.0
        gain_ft = round(gain_m * 3.28084, 1)

        # Pace
        if dist_mi > 0 and dur_s > 0:
            pace_sec = dur_s / dist_mi
            pace_str = f"{int(pace_sec // 60)}:{int(pace_sec % 60):02d}/mi"
            time_str = f"{int(dur_s // 60)}:{int(dur_s % 60):02d}"
        else:
            pace_str = "—"
            time_str = "—"

        raw = {}
        try:
            if d.get("raw_json"):
                raw = json.loads(d["raw_json"])
        except Exception:
            pass

        lat = raw.get("startLatitude") or 39.6079
        lon = raw.get("startLongitude") or -75.7369
        loc = raw.get("locationName") or ""

        # Guess surface
        name_lower = (d.get("name") or "").lower()
        if "xc" in name_lower or "cross" in name_lower or "grass" in name_lower or "trail" in name_lower:
            surface = "xc"
        elif "track" in name_lower:
            surface = "track"
        else:
            surface = "road"

        candidates.append({
            "garmin_id": d["garmin_id"],
            "name": d.get("name") or "Garmin Run",
            "start_time": (d.get("start_time") or "")[:10],
            "distance_m": round(dist_m, 1),
            "distance_miles": round(dist_mi, 2),
            "distance_km": round(dist_km, 2),
            "duration_s": dur_s,
            "time_fmt": time_str,
            "pace_per_mile": pace_str,
            "elevation_gain_ft": gain_ft,
            "lat": lat,
            "lon": lon,
            "location": loc,
            "suggested_surface": surface,
        })
    return candidates


def import_course_from_activity(
    garmin_id: int,
    custom_name: str | None = None,
    surface: str | None = None,
) -> dict:
    """Extract GPS elevation profile and course metadata from a Garmin activity and save as a course."""
    with get_db() as db:
        row = db.execute("SELECT * FROM activities WHERE garmin_id = ?", (garmin_id,)).fetchone()
        activity = dict(row) if row else None

    if not activity:
        raise ValueError(f"Activity with Garmin ID {garmin_id} not found in database.")

    raw = {}
    try:
        if activity.get("raw_json"):
            raw = json.loads(activity["raw_json"])
    except Exception:
        pass

    dist_m = activity.get("distance_m") or 5000.0
    dist_mi = dist_m / 1609.344
    if dist_mi <= 1.05:
        dist_label = "1 Mile"
    elif dist_mi <= 2.1:
        dist_label = "3200m"
    elif dist_mi <= 3.3:
        dist_label = "5K"
    elif dist_mi <= 6.5:
        dist_label = "10K"
    elif dist_mi <= 14.0:
        dist_label = "Half Marathon"
    else:
        dist_label = "Marathon"

    gain_m = activity.get("elevation_gain") or raw.get("elevationGain") or 0.0
    gain_ft = round(gain_m * 3.28084, 1)

    loss_m = activity.get("elevation_loss") or raw.get("elevationLoss") or gain_m
    loss_ft = round(loss_m * 3.28084, 1)

    min_elev_m = raw.get("minElevation") or 15.0
    altitude_ft = round(min_elev_m * 3.28084, 1)

    lat = raw.get("startLatitude") or 39.6079
    lon = raw.get("startLongitude") or -75.7369
    loc_name = raw.get("locationName") or ""

    act_name = activity.get("name") or "Garmin Run"
    course_name = custom_name.strip() if custom_name and custom_name.strip() else f"{act_name} ({dist_label} · +{int(round(gain_ft))} ft / -{int(round(loss_ft))} ft)"
    short_name = custom_name.strip() if custom_name and custom_name.strip() else f"{act_name} (+{int(round(gain_ft))} ft)"

    if not surface:
        name_lower = act_name.lower()
        if "xc" in name_lower or "grass" in name_lower or "trail" in name_lower:
            surface = "xc"
        elif "track" in name_lower:
            surface = "track"
        else:
            surface = "road"

    # Extract 10 elevation profile points and route coordinates
    laps = []
    cache_file = Path.home() / "garmin-coach" / "data" / "garmin_activities.json"
    if cache_file.exists():
        try:
            with open(cache_file, "r", encoding="utf-8") as f:
                cdata = json.load(f)
            for a in cdata.get("activities", []):
                if str(a.get("garmin_activity_id")) == str(garmin_id):
                    laps = a.get("details", {}).get("splits", {}).get("lapDTOs", [])
                    break
        except Exception:
            pass

    profile = []
    route = []
    mile_markers = []

    if laps:
        points = []
        for l in laps:
            min_e = l.get("minElevation")
            max_e = l.get("maxElevation")
            if min_e is not None and max_e is not None:
                mid_e = (min_e + max_e) / 2 * 3.28084
                points.extend([round(min_e * 3.28084, 1), round(mid_e, 1), round(max_e * 3.28084, 1)])
            elif min_e is not None:
                points.append(round(min_e * 3.28084, 1))

            s_lat = l.get("startLatitude")
            s_lon = l.get("startLongitude")
            e_lat = l.get("endLatitude")
            e_lon = l.get("endLongitude")
            if s_lat is not None and s_lon is not None:
                route.append([round(float(s_lat), 6), round(float(s_lon), 6)])
                lap_idx = l.get("lapIndex", len(mile_markers) + 1)
                mile_markers.append({
                    "mile": str(lap_idx),
                    "lat": round(float(s_lat), 6),
                    "lon": round(float(s_lon), 6),
                    "label": f"Mile {lap_idx}"
                })
            if e_lat is not None and e_lon is not None:
                route.append([round(float(e_lat), 6), round(float(e_lon), 6)])

        if len(points) >= 4:
            for i in range(10):
                idx = i * (len(points) - 1) / 9.0
                low = int(idx)
                high = min(low + 1, len(points) - 1)
                frac = idx - low
                val = points[low] * (1 - frac) + points[high] * frac
                profile.append(round(val, 1))

    if not profile or len(profile) < 10:
        multipliers = [0.0, 0.25, 0.65, 0.90, 0.45, 0.80, 1.0, 0.60, 0.30, 0.0]
        profile = [round(altitude_ft + gain_ft * m * 0.7, 1) for m in multipliers]

    if not route:
        # Default synthesized loop around the start coordinate
        route = [
            [lat, lon],
            [lat + 0.003, lon - 0.002],
            [lat + 0.005, lon + 0.003],
            [lat + 0.002, lon + 0.006],
            [lat - 0.002, lon + 0.004],
            [lat, lon],
        ]
        mile_markers = [
            {"mile": "1", "lat": lat + 0.003, "lon": lon - 0.002, "label": "Mile 1"},
            {"mile": "2", "lat": lat + 0.002, "lon": lon + 0.006, "label": "Mile 2"},
            {"mile": "3.1", "lat": lat, "lon": lon, "label": "Finish"}
        ]

    max_inc = round(min(gain_ft / max(dist_mi, 1.0) * 0.05, 12.0), 1)
    max_dec = round(-min(loss_ft / max(dist_mi, 1.0) * 0.05, 12.0), 1)

    course_data = {
        "course_id": f"garmin_{garmin_id}",
        "name": course_name,
        "short_name": short_name,
        "distance_label": dist_label,
        "surface": surface,
        "elevation_gain_ft": gain_ft,
        "elevation_loss_ft": loss_ft,
        "max_incline_pct": max_inc,
        "max_decline_pct": max_dec,
        "altitude_ft": altitude_ft,
        "lat": lat,
        "lon": lon,
        "description": f"Imported GPS profile from Garmin run on {str(activity.get('start_time'))[:10]} in {loc_name or 'Delaware'}. Distance: {dist_mi:.2f} mi.",
        "profile": profile,
        "route": route,
        "mile_markers": mile_markers,
        "is_preset": 0,
        "garmin_activity_id": garmin_id,
    }

    return save_course(course_data)

"""
RaceKit Prediction Engine — Multi-model ensemble race predictor.

Models:
  1. Calibrated Pete Riegel Fatigue Curve — personalized fatigue exponent based on
     the athlete's actual middle-distance to endurance drop-off (not generic 1.06).
  2. Critical Speed (CS) + Anaerobic Work Capacity (D') — Monod & Scherrer hyperbolic
     model fitted strictly on severe-domain efforts (1K to 5K) with sprint velocity bounding.
  3. Jack Daniels' VDOT — oxygen cost and sustainable %VO2max inverted curve.
  4. Real-World Aerobic Volume & Long-Run Durability Constraints (Tanda 2011 / Vickers 2016) —
     prevents dangerous generic extrapolations (e.g. predicting a 3:16 marathon for a runner
     who averages 17-22 mpw with an 8.9 mi longest run).
"""

@dataclass
class Effort:

    distance_m: float
    time_s: float
    date: str
    power_w: Optional[float] = None
    hr_avg: Optional[float] = None
    source: str = "auto"
    label: str = ""

    @property
    def speed_mps(self) -> float:
        return self.distance_m / self.time_s if self.time_s > 0 else 0

    @property
    def pace_per_mile_s(self) -> float:
        return self.time_s / (self.distance_m / 1609.344) if self.distance_m > 0 else 0

@dataclass
class RacePrediction:
    """A single race prediction with confidence bounds and volume durability checks."""
    distance_m: float
    distance_label: str
    time_s: float
    fast_s: float
    slow_s: float
    pace_per_mile_s: float
    model_contributions: dict = field(default_factory=dict)
    adjustments_applied: dict = field(default_factory=dict)
    adjustment_pct: float = 0.0
    theoretical_time_s: float = 0.0
    volume_penalty_pct: float = 0.0
    readiness_status: str = "READY"
    readiness_note: str = ""

    def format_time(self, t: float | None = None) -> str:
        t = t if t is not None else self.time_s
        if t is None or math.isnan(t) or t <= 0:
            return "—"
        if t >= 3600:
            h = int(t // 3600)
            m = int((t % 3600) // 60)
            s = t % 60
            return f"{h}:{m:02d}:{s:04.1f}"
        m = int(t // 60)
        s = t % 60
        return f"{m}:{s:04.1f}"

    def to_dict(self) -> dict:
        return {
            "distance_m": self.distance_m,
            "distance_label": self.distance_label,
            "time_s": round(float(self.time_s), 2),
            "time_fmt": self.format_time(),
            "fast_s": round(float(self.fast_s), 2),
            "fast_fmt": self.format_time(self.fast_s),
            "slow_s": round(float(self.slow_s), 2),
            "slow_fmt": self.format_time(self.slow_s),
            "pace_per_mile": self.format_time(self.pace_per_mile_s),
            "pace_per_mile_s": round(float(self.pace_per_mile_s), 2),
            "theoretical_time_s": round(float(self.theoretical_time_s), 2),
            "theoretical_time_fmt": self.format_time(self.theoretical_time_s),
            "volume_penalty_pct": round(float(self.volume_penalty_pct), 1),
            "readiness_status": self.readiness_status,
            "readiness_note": self.readiness_note,
            "models": self.model_contributions,
            "adjustments": self.adjustments_applied,
            "adjustment_pct": round(float(self.adjustment_pct), 2),
        }

# Prediction Engine

class PredictionEngine:
    """
    Multi-model race prediction engine.
    Calibrated against verified athlete efforts, physiology, and actual training volume.
    """

    def __init__(
        self,
        efforts: list[Effort],
        vo2max: Optional[float] = None,
        recency_halflife_days: float = 60.0,
        volume: Optional[VolumeProfile] = None,
    ):
        self.efforts = sorted(efforts, key=lambda e: e.distance_m)
        self.vo2max = vo2max
        self.recency_halflife = recency_halflife_days
        self.volume = volume or VolumeProfile()

        # Fitted parameters
        self.cs: Optional[float] = None
        self.d_prime: Optional[float] = None
        self.cs_r2: Optional[float] = None
        self.b_mile_to_5k: Optional[float] = None
        self.mile_effort: Optional[Effort] = None
        self.five_k_effort: Optional[Effort] = None
        self.effective_vdot: Optional[float] = None

        self._fit_models()

    def _recency_weight(self, effort_date_str: str) -> float:
        try:
            ed = datetime.fromisoformat(effort_date_str[:10])
            days_ago = (datetime.now() - ed).days
            return 2.0 ** (-max(days_ago, 0) / self.recency_halflife)
        except Exception:
            return 0.5

    def _fit_models(self):
        if not self.efforts:
            return

        # 1. Identify key anchor efforts (Mile & 5K if present)
        for e in self.efforts:
            if abs(e.distance_m - 1609.344) < 100 or "mile" in e.label.lower():
                if self.mile_effort is None or e.time_s < self.mile_effort.time_s:
                    self.mile_effort = e
            elif abs(e.distance_m - 5000.0) < 100 or "5k" in e.label.lower():
                if self.five_k_effort is None or e.time_s < self.five_k_effort.time_s:
                    self.five_k_effort = e

        # Calculate empirical fatigue exponent between Mile and 5K
        if self.mile_effort and self.five_k_effort:
            ratio_dist = self.five_k_effort.distance_m / self.mile_effort.distance_m
            ratio_time = self.five_k_effort.time_s / self.mile_effort.time_s
            try:
                self.b_mile_to_5k = math.log(ratio_time) / math.log(ratio_dist)
            except Exception:
                self.b_mile_to_5k = 1.10
        elif len(self.efforts) >= 2:
            e1 = self.efforts[0]
            e2 = self.efforts[-1]
            try:
                self.b_mile_to_5k = math.log(e2.time_s / e1.time_s) / math.log(e2.distance_m / e1.distance_m)
            except Exception:
                self.b_mile_to_5k = 1.08
        else:
            self.b_mile_to_5k = 1.08

        # 2. Fit Critical Speed (CS) and D'
        # Include efforts up to 10K (12,000m / 3600s) to anchor sustainable threshold pace
        cs_efforts = [
            e for e in self.efforts
            if 800 <= e.distance_m <= 12000 and 120 <= e.time_s <= 3600
        ]
        if len(cs_efforts) < 2:
            cs_efforts = self.efforts[:min(len(self.efforts), 4)]

        if len(cs_efforts) >= 2:
            times = [e.time_s for e in cs_efforts]
            dists = [e.distance_m for e in cs_efforts]
            weights = [self._recency_weight(e.date) for e in cs_efforts]

            w_sum = sum(weights)
            t_bar = sum(w * t for w, t in zip(weights, times)) / w_sum
            d_bar = sum(w * d for w, d in zip(weights, dists)) / w_sum

            cov_td = sum(w * (t - t_bar) * (d - d_bar) for w, t, d in zip(weights, times, dists))
            var_t = sum(w * (t - t_bar) ** 2 for w, t in zip(weights, times))

            if var_t > 0:
                cs = cov_td / var_t
                d_prime = d_bar - cs * t_bar

                # Physiological safeguard: Critical Speed is maximal metabolic steady state (MMSS).
                # For runners without high marathon volume (< 25 mpw), CS represents a pace
                # holdable for 45-60 min (near 10K/threshold pace), capped at ~86% of 5K speed.
                if self.five_k_effort:
                    v_5k = self.five_k_effort.distance_m / self.five_k_effort.time_s
                    max_sustainable_cs = v_5k * (0.86 if self.volume.avg_weekly_miles < 25 else 0.90)
                    if cs > max_sustainable_cs:
                        cs = max_sustainable_cs
                        if self.mile_effort:
                            d_prime = max(50.0, self.mile_effort.distance_m - cs * self.mile_effort.time_s)

                if 2.5 < cs < 6.0 and 50 < d_prime < 800:
                    self.cs = float(cs)
                    self.d_prime = float(d_prime)

                    ss_tot = sum(w * (d - d_bar) ** 2 for w, d in zip(weights, dists))
                    ss_res = sum(w * (d - (cs * t + d_prime)) ** 2 for w, t, d in zip(weights, times, dists))
                    self.cs_r2 = float(1.0 - (ss_res / ss_tot)) if ss_tot > 0 else 0.99

        # 3. Calculate effective VDOT from performances
        vdot_candidates = []
        for e in self.efforts:
            vdot = self._vdot_from_performance(e.distance_m, e.time_s)
            if vdot:
                vdot_candidates.append(vdot)
        if vdot_candidates:
            self.effective_vdot = float(max(vdot_candidates))
        elif self.vo2max:
            self.effective_vdot = float(self.vo2max)

    # ── Jack Daniels' VDOT formulas ───────────────────────────────

    def _vdot_from_performance(self, dist_m: float, time_s: float) -> Optional[float]:
        t_min = time_s / 60.0
        if t_min <= 0:
            return None
        v_m_min = dist_m / t_min
        vo2 = -4.60 + 0.182258 * v_m_min + 0.000104 * (v_m_min ** 2)
        pct_max = 0.8 + 0.1894393 * math.exp(-0.012778 * t_min) + 0.2989558 * math.exp(-0.1932605 * t_min)
        if pct_max <= 0:
            return None
        return vo2 / pct_max

    def _predict_daniels(self, dist_m: float) -> Optional[float]:
        vdot = self.effective_vdot
        if not vdot:
            return None

        low = 30.0
        high = 20000.0
        for _ in range(50):
            mid = (low + high) / 2.0
            v_est = self._vdot_from_performance(dist_m, mid)
            if v_est is None:
                break
            if v_est < vdot:
                high = mid
            else:
                low = mid
        return float(mid)

    # ── Calibrated Pete Riegel Model ───────────────────────────────

    def _predict_riegel(self, dist_m: float) -> Optional[float]:
        base_effort = self.mile_effort or (self.efforts[0] if self.efforts else None)
        if not base_effort:
            return None

        b_base = self.b_mile_to_5k or 1.10

        if dist_m <= 800:
            b = 1.082
            ref_dist = base_effort.distance_m
            ref_time = base_effort.time_s
        elif dist_m <= 1609.344:
            b = 1.060
            ref_dist = base_effort.distance_m
            ref_time = base_effort.time_s
        elif dist_m <= 3200:
            b = 1.100
            ref_dist = base_effort.distance_m
            ref_time = base_effort.time_s
        elif dist_m <= 5000:
            b = b_base * 0.985
            ref_dist = base_effort.distance_m
            ref_time = base_effort.time_s
        else:
            b = 1.140
            if self.five_k_effort:
                ref_dist = self.five_k_effort.distance_m
                ref_time = self.five_k_effort.time_s
            else:
                ref_dist = base_effort.distance_m
                ref_time = base_effort.time_s

        pred = ref_time * ((dist_m / ref_dist) ** b)
        return float(pred)

    # ── Critical Speed Model ───────────────────────────────────────

    def _predict_cs(self, dist_m: float) -> Optional[float]:
        if self.cs is None or self.d_prime is None:
            return None

        if dist_m <= self.d_prime:
            return float(dist_m / (self.cs * 1.35))
        elif dist_m <= 1000:
            v = self.cs * (1.0 + (self.d_prime / dist_m))
            v = min(v, self.cs * 1.32)
            return float(dist_m / v)
        else:
            t = (dist_m - self.d_prime) / self.cs
            return float(t) if t > 0 else None

    # ── Ensemble Predictor with Volume & Durability Checks ─────────

    def predict(self, distance_m: float, label: str = "") -> RacePrediction:
        models: dict[str, float] = {}

        riegel = self._predict_riegel(distance_m)
        if riegel is not None and riegel > 5:
            models["calibrated_riegel"] = riegel

        cs = self._predict_cs(distance_m)
        if cs is not None and cs > 5:
            models["critical_speed"] = cs

        daniels = self._predict_daniels(distance_m)
        if daniels is not None and daniels > 5:
            models["daniels_vdot"] = daniels

        if not models:
            if self.efforts:
                best = min(self.efforts, key=lambda e: e.time_s / e.distance_m)
                fallback = best.time_s * (distance_m / best.distance_m) ** 1.06
                models["standard_riegel"] = fallback
            else:
                models["standard_riegel"] = 300.0 * (distance_m / 1600.0)

        # ── Exact 1600m / 1 Mile Coupling ────────────────────────────
        # Strictly scale 1600m from 1 Mile (1609.344m) by distance ratio (1600 / 1609.344)
        if (abs(distance_m - 1600.0) < 1.0 or label == "1600m") and abs(distance_m - 1609.344) > 1.0:
            mile_pred = self.predict(1609.344, label="Mile")
            ratio_1600 = 1600.0 / 1609.344
            
            t_1600 = mile_pred.time_s * ratio_1600
            fast_1600 = mile_pred.fast_s * ratio_1600
            slow_1600 = mile_pred.slow_s * ratio_1600
            theo_1600 = mile_pred.theoretical_time_s * ratio_1600
            
            contributions = {}
            dummy_pred = RacePrediction(1600.0, "", 0, 0, 0, 0)
            for name, m in mile_pred.model_contributions.items():
                if name.startswith("_"):
                    continue
                t_sub = m["time_s"] * ratio_1600
                contributions[name] = {
                    "time_s": round(float(t_sub), 2),
                    "time_fmt": dummy_pred.format_time(t_sub),
                    "weight": m["weight"],
                }
            contributions["_summary"] = {
                "theoretical_time_s": round(float(theo_1600), 2),
                "theoretical_time_fmt": dummy_pred.format_time(theo_1600),
                "volume_penalty_pct": mile_pred.volume_penalty_pct,
                "base_time_s": round(float(t_1600), 2),
                "base_time_fmt": dummy_pred.format_time(t_1600),
            }

            return RacePrediction(
                distance_m=1600.0,
                distance_label="1600m",
                time_s=float(t_1600),
                fast_s=float(fast_1600),
                slow_s=float(slow_1600),
                pace_per_mile_s=float(mile_pred.pace_per_mile_s),
                model_contributions=contributions,
                theoretical_time_s=float(theo_1600),
                volume_penalty_pct=float(mile_pred.volume_penalty_pct),
                readiness_status=mile_pred.readiness_status,
                readiness_note=mile_pred.readiness_note,
            )

        # Distance-specific physiological ensemble weights
        if distance_m <= 850:
            weights = {"calibrated_riegel": 0.50, "critical_speed": 0.30, "daniels_vdot": 0.20}
        elif distance_m <= 1650:
            weights = {"calibrated_riegel": 0.45, "critical_speed": 0.35, "daniels_vdot": 0.20}
        elif distance_m <= 5500:
            weights = {"calibrated_riegel": 0.35, "critical_speed": 0.30, "daniels_vdot": 0.35}
        else:
            weights = {"calibrated_riegel": 0.45, "critical_speed": 0.10, "daniels_vdot": 0.45}

        total_w = sum(weights.get(k, 0.2) for k in models)
        theoretical_s = sum(models[k] * weights.get(k, 0.2) for k in models) / total_w

        # ── Realistic Training Volume & Long-Run Constraint Engine ───
        dist_mi = distance_m / 1609.344
        longest_mi = self.volume.longest_run_miles
        weekly_mi = self.volume.avg_weekly_miles

        ratio = dist_mi / max(longest_mi, 1.0)
        long_penalty = 0.0
        if ratio > 1.05:
            # Non-linear penalty when race distance exceeds longest run
            long_penalty = min(40.0, 14.0 * ((ratio - 1.0) ** 1.35))

        target_mpw = 15.0 if dist_mi <= 3.1 else (25.0 if dist_mi <= 6.2 else (35.0 if dist_mi <= 13.1 else 50.0))
        vol_penalty = 0.0
        if weekly_mi < target_mpw and dist_mi > 5.0:
            vol_penalty = min(25.0, ((target_mpw - weekly_mi) / target_mpw) * 22.0 * (dist_mi / 26.2))

        volume_penalty_pct = round(long_penalty + vol_penalty, 1)

        # Realistic volume-constrained time
        realistic_s = theoretical_s * (1.0 + volume_penalty_pct / 100.0)

        # Training feasibility evaluation
        if ratio <= 1.05 and dist_mi <= 6.2:
            readiness_status = "READY"
            readiness_note = "Fully supported by your weekly volume and long run base."
        elif ratio <= 1.6 and dist_mi <= 14.0:
            readiness_status = "VOLUME STRETCH"
            readiness_note = f"Distance ({dist_mi:.1f} mi) exceeds your {longest_mi:.1f} mi long run. Aerobic fade expected."
        else:
            readiness_status = "HIGH BONK RISK"
            readiness_note = (
                f"Severe aerobic base deficit: longest run is only {longest_mi:.1f} mi and weekly volume is {weekly_mi:.0f} mpw "
                f"(vs {target_mpw:.0f}+ mpw recommended). Glycogen exhaustion and eccentric muscle breakdown will force severe pace collapse."
            )

        # Spread & confidence bounds
        vals = list(models.values())
        if len(vals) > 1:
            spread = (max(vals) - min(vals)) / theoretical_s
            margin = max(min(spread * 0.45, 0.035), 0.012)
        else:
            margin = 0.02

        fast = realistic_s * (1.0 - margin)
        slow = realistic_s * (1.0 + margin)
        pace = realistic_s / dist_mi

        if not label:
            closest = min(STANDARD_DISTANCES.items(), key=lambda x: abs(x[1] - distance_m))
            label = closest[0] if abs(closest[1] - distance_m) < 40 else f"{int(distance_m)}m"

        contributions = {}
        dummy_pred = RacePrediction(distance_m, "", 0, 0, 0, 0)
        for name, t in models.items():
            contributions[name] = {
                "time_s": round(float(t), 2),
                "time_fmt": dummy_pred.format_time(t),
                "weight": round(float(weights.get(name, 0.2) / total_w), 3),
            }
        contributions["_summary"] = {
            "theoretical_time_s": round(float(theoretical_s), 2),
            "theoretical_time_fmt": dummy_pred.format_time(theoretical_s),
            "volume_penalty_pct": float(volume_penalty_pct),
            "base_time_s": round(float(realistic_s), 2),
            "base_time_fmt": dummy_pred.format_time(realistic_s),
        }

        return RacePrediction(
            distance_m=distance_m,
            distance_label=label,
            time_s=float(realistic_s),
            fast_s=float(fast),
            slow_s=float(slow),
            pace_per_mile_s=float(pace),
            model_contributions=contributions,
            theoretical_time_s=float(theoretical_s),
            volume_penalty_pct=float(volume_penalty_pct),
            readiness_status=readiness_status,
            readiness_note=readiness_note,
        )

    def predict_all(self, distances: list[str] | None = None) -> list[RacePrediction]:
        if distances is None:
            distances = PRIMARY_DISTANCES
        results = []
        for name in distances:
            d = STANDARD_DISTANCES.get(name)
            if d:
                results.append(self.predict(d, label=name))
        return results

    def get_fitness_metrics(self) -> FitnessMetrics:
        m = FitnessMetrics(
            critical_speed_mps=self.cs,
            d_prime_m=self.d_prime,
            riegel_exponent=self.b_mile_to_5k,
            vo2max=self.effective_vdot or self.vo2max,
            num_efforts=len(self.efforts),
            model_fit_r2=self.cs_r2,
            longest_run_miles=self.volume.longest_run_miles,
            avg_weekly_miles=self.volume.avg_weekly_miles,
        )

        if self.cs:
            pace_s = 1609.344 / self.cs
            mins = int(pace_s // 60)
            secs = pace_s % 60
            m.critical_speed_pace = f"{mins}:{secs:04.1f}/mi"

        if self.b_mile_to_5k:
            if self.b_mile_to_5k < 1.05:
                m.fatigue_profile = "Endurance Specialist"
            elif self.b_mile_to_5k < 1.09:
                m.fatigue_profile = "Balanced Distance"
            elif self.b_mile_to_5k < 1.14:
                m.fatigue_profile = "Middle-Distance / Fast-Twitch"
            else:
                m.fatigue_profile = "High Anaerobic Bias"

        return m

# ── Dynamic Physiological Training Paces (Daniels / CS) ───────

    @staticmethod
    def _fmt_pace(sec_per_mile: float) -> str:
        m = int(sec_per_mile // 60)
        s = int(round(sec_per_mile % 60))
        if s == 60:
            m += 1
            s = 0
        return f"{m}:{s:02d}/mi"

    def _pace_at_vo2_pct(self, pct: float) -> Optional[float]:
        vdot = self.effective_vdot or self.vo2max
        if not vdot or vdot <= 0:
            return None
        target_vo2 = vdot * pct
        # Solve 0.000104*v^2 + 0.182258*v - (4.60 + target_vo2) = 0 (m/min)
        disc = (0.182258 ** 2) + 4.0 * 0.000104 * (4.60 + target_vo2)
        if disc <= 0:
            return None
        v_m_min = (-0.182258 + math.sqrt(disc)) / (2.0 * 0.000104)
        v_mps = v_m_min / 60.0
        return 1609.344 / v_mps if v_mps > 0 else None

    def get_training_zones(self) -> dict:
        """Return personalized pace zones derived from live VDOT and Critical Speed."""
        easy_slow = self._pace_at_vo2_pct(0.64) or 520.0
        easy_fast = self._pace_at_vo2_pct(0.71) or 485.0
        marathon = self._pace_at_vo2_pct(0.81) or 435.0
        threshold_vdot = self._pace_at_vo2_pct(0.88) or 395.0
        cs_pace = (1609.344 / self.cs) if self.cs else threshold_vdot
        # Blend Critical Speed and Daniels T-pace for threshold durability
        threshold = (threshold_vdot + min(cs_pace, threshold_vdot + 18.0)) / 2.0
        interval = self._pace_at_vo2_pct(0.98) or 350.0
        rep = self._pace_at_vo2_pct(1.07) or 320.0

        return {
            "recovery": {"pace_s": round(easy_slow + 20, 1), "label": f"{self._fmt_pace(easy_slow + 10)}–{self._fmt_pace(easy_slow + 35)}"},
            "easy": {"pace_s": round((easy_slow + easy_fast) / 2, 1), "label": f"{self._fmt_pace(easy_fast)}–{self._fmt_pace(easy_slow)}"},
            "marathon": {"pace_s": round(marathon, 1), "label": self._fmt_pace(marathon)},
            "threshold": {"pace_s": round(threshold, 1), "label": f"{self._fmt_pace(threshold - 4)}–{self._fmt_pace(threshold + 4)}"},
            "interval": {"pace_s": round(interval, 1), "label": self._fmt_pace(interval)},
            "repetition": {"pace_s": round(rep, 1), "label": self._fmt_pace(rep)},
            "rep_400m": f"{int((rep / 4.02336) // 60)}:{(rep / 4.02336) % 60:04.1f}",
        }

    # ── Course-Aware Race Splits Generator ────────────────────────

    @staticmethod
    def generate_splits(
        prediction: RacePrediction,
        strategy: str = "course_gap",
        course_profile_ft: Optional[list[float]] = None,
    ) -> list[dict]:
        """
        Break an adjusted RacePrediction into actionable pacing splits.
        Strategies:
          - 'even': Equal effort / equal pace per segment
          - 'negative': Controlled opening (+1.8%), steady middle, fast closing kick (-2.2%)
          - 'course_gap': Grade-Adjusted Pace using the course's elevation profile
        """
        dist_m = prediction.distance_m
        total_s = prediction.time_s
        if dist_m <= 0 or total_s <= 0:
            return []

        # Build natural checkpoints (400m laps for <=1650m; 800m for <=3250m; 1-mile splits for >=5K)
        checkpoints: list[tuple[str, float]] = []
        if dist_m <= 1650:
            step = 400.0
            cur = 0.0
            lap = 1
            while cur + step < dist_m - 15:
                cur += step
                checkpoints.append((f"Lap {lap} ({int(cur)}m)", step))
                lap += 1
            rem = dist_m - cur
            if rem > 1:
                checkpoints.append((f"Finish ({int(round(dist_m))}m)", rem))
        elif dist_m <= 3250:
            step = 800.0
            cur = 0.0
            lap = 1
            while cur + step < dist_m - 25:
                cur += step
                checkpoints.append((f"{int(cur)}m", step))
                lap += 1
            rem = dist_m - cur
            if rem > 1:
                checkpoints.append((f"Finish ({int(round(dist_m))}m)", rem))
        else:
            mile_m = 1609.344
            cur = 0.0
            mi = 1
            while cur + mile_m < dist_m - 60:
                cur += mile_m
                checkpoints.append((f"Mile {mi}", mile_m))
                mi += 1
            rem = dist_m - cur
            if rem > 1:
                rem_mi = rem / mile_m
                label = f"Kick ({rem_mi:.2f} mi)" if rem_mi < 0.9 else f"Mile {mi}"
                checkpoints.append((label, rem))

        def _interp_elev(frac: float) -> float:
            if not course_profile_ft or len(course_profile_ft) < 2:
                return 0.0
            pos = max(0.0, min(1.0, frac)) * (len(course_profile_ft) - 1)
            lo = int(pos)
            hi = min(lo + 1, len(course_profile_ft) - 1)
            f = pos - lo
            return course_profile_ft[lo] * (1.0 - f) + course_profile_ft[hi] * f

        n = len(checkpoints)
        avg_pace_mps = dist_m / total_s
        raw_times: list[float] = []
        seg_meta: list[dict] = []

        cum_dist = 0.0
        for idx, (lbl, seg_d) in enumerate(checkpoints):
            start_frac = cum_dist / dist_m
            end_frac = (cum_dist + seg_d) / dist_m
            cum_dist += seg_d

            elev_start = _interp_elev(start_frac)
            elev_end = _interp_elev(end_frac)
            delta_ft = elev_end - elev_start
            delta_m = delta_ft * 0.3048
            grade_pct = (delta_m / seg_d) * 100.0 if seg_d > 0 else 0.0

            factor = 1.0
            cue = "Hold steady rhythm"

            if strategy == "negative":
                # Smooth linear progression from +1.8% opening to -2.2% finish
                prog = idx / max(n - 1, 1)
                factor = 1.018 - 0.040 * prog
                cue = "Controlled patience" if prog < 0.35 else ("Lock into rhythm" if prog < 0.7 else "Empty the tank")
            elif strategy == "course_gap":
                # Terrain adjustment + gentle -1.0% finish kick
                if grade_pct >= 0:
                    grade_adj = grade_pct * 0.034
                else:
                    grade_adj = max(grade_pct * 0.018, -0.05)
                kick_adj = -0.012 if idx == n - 1 and n >= 3 else (0.008 if idx == 0 else 0.0)
                factor = max(0.90, 1.0 + grade_adj + kick_adj)
                if delta_ft >= 12:
                    cue = f"Climb (+{int(round(delta_ft))} ft) · Shorten stride"
                elif delta_ft <= -12:
                    cue = f"Descent ({int(round(delta_ft))} ft) · Open turnover"
                elif idx == n - 1:
                    cue = "Final kick · Empty D' reserve"
                elif idx == 0:
                    cue = "Controlled surge · Avoid early redline"
                else:
                    cue = "Aerobic lock · Smooth cadence"

            raw_times.append((seg_d / avg_pace_mps) * factor)
            seg_meta.append({
                "label": lbl,
                "segment_m": round(seg_d, 1),
                "elev_delta_ft": round(delta_ft, 1),
                "grade_pct": round(grade_pct, 1),
                "cue": cue,
            })

        # Normalize so splits sum identically to total_s
        scale = total_s / sum(raw_times)
        splits = []
        cum_t = 0.0
        dummy = RacePrediction(dist_m, "", 0, 0, 0, 0)

        for rt, meta in zip(raw_times, seg_meta):
            seg_t = rt * scale
            cum_t += seg_t
            pace_s = seg_t / (meta["segment_m"] / 1609.344)
            splits.append({
                "label": meta["label"],
                "segment_time_s": round(seg_t, 2),
                "segment_time_fmt": dummy.format_time(seg_t),
                "cumulative_time_s": round(cum_t, 2),
                "cumulative_time_fmt": dummy.format_time(cum_t),
                "pace_per_mile": dummy.format_time(pace_s),
                "elev_delta_ft": meta["elev_delta_ft"],
                "grade_pct": meta["grade_pct"],
                "cue": meta["cue"],
            })

        return splits

  Surface      — Pinnington & Dawson 2001, Frederick 1986
  Shoes        — Hoogkamer et al. 2018 (Vaporfly study), Barnes & Kilding 2015

# Individual adjustment functions — each returns % change
# (positive = slower, negative = faster)
# ═══════════════════════════════════════════════════════════════════

def _temperature_adjustment(temp_f: float, distance_m: float) -> float:
    """
    Ely et al. 2007, Galloway & Maughan 1997:
    Performance degrades above ~50°F.
    Severe non-linear cardiovascular drift as temp exceeds 80°F - 100°F.
    """
    optimal = 48.0  # °F — research optimal for distance running
    if temp_f <= optimal:
        # Cold has minimal effect until extreme
        if temp_f < 20:
            return (20 - temp_f) * 0.1  # Slight penalty for extreme cold
        return 0.0

    excess = temp_f - optimal
    # Base penalty per degree above optimal
    if distance_m <= 1650:
        base_rate = 0.10
    elif distance_m <= 5000:
        base_rate = 0.16
    elif distance_m <= 10000:
        base_rate = 0.22
    elif distance_m <= 21097:
        base_rate = 0.30
    else:
        base_rate = 0.40

    linear_penalty = excess * base_rate
    # Severe thermal strain non-linearity above 80°F (ambient air approaching core temp)
    extreme_heat = max(0.0, temp_f - 80.0)
    non_linear_penalty = 0.020 * (extreme_heat ** 2) * (1.0 if distance_m <= 5000 else 1.2)

    return linear_penalty + non_linear_penalty


def _humidity_adjustment(humidity_pct: float, temp_f: float) -> float:
    """
    Humidity impairs evaporative sweat cooling (Montain et al. 2007).
    Becomes catastrophic above 80°F + 70% humidity (wet bulb approach).
    At 100°F + 100% humidity, humidity alone adds ~12-16% penalty (total heat stress ~25-32%).
    """
    if temp_f < 55 or humidity_pct < 50:
        return 0.0

    excess_humid = humidity_pct - 50.0
    excess_temp = temp_f - 55.0

    # Progressive interaction
    base_inter = excess_humid * excess_temp * 0.0028

    # Exponential sweat evaporation breakdown above 80°F and 70% RH
    if temp_f > 80.0 and humidity_pct > 70.0:
        heat_index_surge = ((temp_f - 80.0) / 20.0) * ((humidity_pct - 70.0) / 30.0) * 8.0
        return base_inter + heat_index_surge

    return base_inter


def _wind_adjustment(wind_mph: float, direction: str) -> float:
    """
    Davies 1980 / Pugh 1971 aerodynamic drag model.
    Aerodynamic power scales quadratically with relative airspeed.
    Headwind at 40 mph penalizes ~12–16%.
    """
    if wind_mph < 2:
        return 0.0

    d = direction.lower()
    if d == "head":
        # Quadratic drag curve: 40 mph -> ~14.4% penalty
        return 0.22 * wind_mph + 0.0035 * (wind_mph ** 2)
    elif d == "tail":
        # Tailwind benefit diminishes due to stride turnover limit
        raw_benefit = -(0.08 * wind_mph - 0.0006 * (wind_mph ** 2))
        return max(raw_benefit, -3.8)
    elif d == "cross":
        # Crosswind induces stabilizing lateral resistance
        return 0.08 * wind_mph + 0.0012 * (wind_mph ** 2)
    else:  # mixed / variable out-and-back
        # Headwind penalty always exceeds tailwind savings
        return 0.07 * wind_mph + 0.0018 * (wind_mph ** 2)


def _elevation_adjustment(gain_m: float, loss_m: Optional[float], distance_m: float) -> float:
    """
    Minetti et al. 2002; Davies 1980; Jack Daniels Running Formula.
    Accounts for the metabolic cost of inclines (climbing) AND the partial recovery
    plus eccentric braking penalties on subsequent declines (descents).

    Biomechanical Principles:
    1. Incline Phase: Uphill running costs ~3.6% extra energy per 100m of climb (approx 12-14s/100ft in 5K).
    2. Decline Phase: Downhill running recovers ~50-55% of climbing time (-1.9% per 100m / approx 6-7s/100ft).
       Downhill cannot fully compensate because stride rate hits neuromuscular ceilings and braking forces increase.
    3. Severe / Technical Declines: When descent rate exceeds ~25m/km, eccentric quadriceps loading causes
       braking deceleration and leg fatigue that carries into subsequent flat miles.
    """
    if distance_m <= 0:
        return 0.0

    if loss_m is None:
        loss_m = gain_m  # Assume loop or out-and-back course with matching decline if omitted

    # Scale factor relative to 5K (shorter races suffer more per meter of hill; longer absorb slightly better)
    distance_factor = (5000.0 / distance_m) ** 0.4 if distance_m > 0 else 1.0

    # Incline climb penalty (% slowdown)
    climb_penalty = (gain_m / 100.0) * 3.6

    # Decline descent recovery (% speedup)
    descent_recovery = (loss_m / 100.0) * 1.9

    # Eccentric downhill braking penalty on steep courses
    loss_per_km = (loss_m / (distance_m / 1000.0)) if distance_m > 0 else 0.0
    braking_penalty = 0.0
    if loss_per_km > 25.0:
        # Steep descent fatigue (e.g. Cemetery Hill descent at Brandywine Creek)
        braking_penalty = min((loss_per_km - 25.0) * 0.02, 1.2)

    net_percent = (climb_penalty - descent_recovery + braking_penalty) * distance_factor

    # Biomechanical limit: net downhill cannot speed up beyond -4.5% due to neuromuscular turnover ceilings
    return max(net_percent, -4.5)


def _surface_adjustment(surface: str) -> float:
    """
    Surface energy cost penalties relative to road.
    Pinnington & Dawson 2001; Frederick 1986.
    Calibrated with Ephraim's verified 19:29 XC 5K benchmark.
    """
    penalties = {
        "track": -0.5,        # Slightly faster than road (sprung surface, banked/cambered)
        "road": 0.0,          # Baseline asphalt/pavement
        "treadmill": -1.0,    # No wind resistance, belt assistance
        "grass": 3.0,         # Manicured park grass
        "xc": 3.6,            # Cross country: grass + uneven turf + turns (anchors 19:29 PR)
        "trail": 4.5,         # Variable dirt terrain, footing roots
        "gravel": 2.5,        # Packed crushed gravel / stone dust
        "sand": 14.0,         # Loose energy-sapping sand
    }
    return penalties.get(surface.lower(), 0.0)


def _shoe_adjustment(shoe_type: str, surface: str = "road", distance_m: float = 5000) -> float:
    """
    Shoe economy adjustment based on shoe technology and surface interaction.
    Exact 5-Shoe Rotation:
      1. Nike Dragonfly XC (XC Spikes)
      2. Nike Vaporfly Next% 2 (Carbon Road Super-Shoe)
      3. Nike Dragonfly 2 (Track Spikes)
      4. Nike Streakfly (Lightweight Racing Flat / Workout Shoe)
      5. New Balance 1080 v15 (Daily Trainer)
    """
    shoe = shoe_type.lower()
    surf = surface.lower()

    # 1. Track surface dynamics
    if surf == "track":
        if shoe in ("dragonfly_track", "dragonfly_2", "dragonfly", "track_super_spikes", "spikes"):
            # Pure track weapon: ZoomX + 135g featherweight + 6-pin track bite (-3.5%)
            return -3.5
        elif shoe in ("dragonfly_xc", "xc_super_spikes"):
            # XC spike on track: ZoomX + pins, slight rubber outsole weight penalty (-2.8%)
            return -2.8
        elif shoe in ("streakfly", "race_flat"):
            # Featherweight 170g ZoomX flat: agile track curve handling (-2.6%)
            return -2.6
        elif shoe in ("vaporfly_next2", "carbon_racer", "super_shoe", "vaporfly"):
            # 40mm road shoes on track curves suffer ankle torque instability & +60g mass (-1.8%)
            return -1.8
        elif shoe in ("nb_1080", "daily_trainer", "trainer"):
            # Max-cushion 300g trainer weight penalty (+1.6%)
            return 1.6
        else:
            return 0.0

    # 2. Cross-Country / Grass surface dynamics
    elif surf in ("grass", "xc", "cross_country"):
        if shoe in ("dragonfly_xc", "xc_super_spikes"):
            # ZoomX + 6 steel spike pins clawing turf (-3.9%)
            # Net on XC: 19:14.2 (clean sub-19:20 speed improvement over Streakfly)
            return -3.9
        elif shoe in ("streakfly", "race_flat"):
            # Lightweight ZoomX flat without spikes (-2.6%)
            # Net on XC: EXACT 19:29.0 verified XC 5K PR anchor
            return -2.6
        elif shoe in ("dragonfly_track", "dragonfly_2", "dragonfly"):
            # Track spike on grass: pins provide traction, but lacks mud-shedding rubber outsole (-3.3%)
            return -3.3
        elif shoe in ("nb_1080", "daily_trainer", "trainer"):
            # 300g daily trainer on grass: zero spike traction, turf slippage (-0.7%)
            return -0.7
        elif shoe in ("vaporfly_next2", "carbon_racer", "super_shoe", "vaporfly"):
            # 40mm stack height wobbles on uneven grass; road rubber slips (0.0%)
            return 0.0
        elif shoe in ("trail_shoe",):
            return -1.5
        else:
            return 0.0

    # 3. Trail surface dynamics
    elif surf in ("trail", "gravel"):
        if shoe in ("dragonfly_xc", "xc_super_spikes"):
            return -2.0
        elif shoe in ("streakfly", "race_flat"):
            return -0.5
        elif shoe in ("nb_1080", "daily_trainer", "trainer"):
            return 0.8
        elif shoe in ("vaporfly_next2", "carbon_racer", "vaporfly"):
            return 1.5
        elif shoe in ("dragonfly_track", "dragonfly_2"):
            return 6.0
        else:
            return 0.0

    # 4. Road surface dynamics (default asphalt)
    else:
        if shoe in ("streakfly", "race_flat"):
            # Featherweight 170g ZoomX road flat: optimal 5K cadence & acceleration (-3.0%)
            return -3.0
        elif shoe in ("vaporfly_next2", "carbon_racer", "super_shoe", "vaporfly"):
            # 40mm ZoomX marathon racer: heavier stack & stiff plate reduce 5K cadence turnover compared to Streakfly (-2.0%)
            return -2.0
        elif shoe in ("nb_1080", "daily_trainer", "trainer"):
            # Max-cushion 300g trainer weight penalty vs racing shoe (+1.6%)
            return 1.6
        elif shoe in ("dragonfly_xc", "xc_super_spikes"):
            # Metal pins on road asphalt/concrete slip and ruin pin threads (+8.0%)
            return 8.0
        elif shoe in ("dragonfly_track", "dragonfly_2", "dragonfly", "spikes"):
            # Track pins on asphalt are hazardous (+10.0%)
            return 10.0
        else:
            return 0.0


def _altitude_adjustment(altitude_m: float, distance_m: float) -> float:
    """
    Péronnet et al. 1991 / Bassett et al. 1999.
    VO2max decreases ~6-7% per 1000m above ~1500m.
    Effect is larger for longer aerobic events.
    """
    if altitude_m < 500:
        return 0.0

    # Progressive penalty above 500m
    if altitude_m < 1500:
        penalty_per_1000 = 2.0
    elif altitude_m < 2500:
        penalty_per_1000 = 4.0
    else:
        penalty_per_1000 = 6.5

    base = (altitude_m - 500) / 1000 * penalty_per_1000

    # Longer races affected more
    distance_factor = 1.0
    if distance_m > 10000:
        distance_factor = 1.2
    elif distance_m > 21000:
        distance_factor = 1.4

    return base * distance_factor


def _readiness_adjustment(
    hrv_status: str = "balanced",
    sleep_score: Optional[int] = None,
    body_battery: Optional[int] = None,
) -> float:
    """
    Race-day readiness modifier based on Garmin physiological data.
    Plews et al. 2013: HRV is a reliable marker of readiness.
    """
    total = 0.0

    # HRV status
    hrv_penalties = {
        "balanced": 0.0,
        "low": 1.0,
        "unbalanced": 1.5,
        "poor": 2.5,
    }
    total += hrv_penalties.get(hrv_status.lower(), 0.0)

    # Sleep score
    if sleep_score is not None:
        if sleep_score >= 80:
            total -= 0.3  # Good sleep = slight boost
        elif sleep_score >= 60:
            total += 0.0  # Adequate
        elif sleep_score >= 40:
            total += 1.0  # Poor sleep
        else:
            total += 2.0  # Very poor sleep

    # Body battery
    if body_battery is not None:
        if body_battery >= 80:
            total -= 0.5  # Fully charged
        elif body_battery >= 50:
            total += 0.0  # Normal
        elif body_battery >= 25:
            total += 1.0  # Low
        else:
            total += 2.5  # Depleted

    return total

def apply_conditions(
    prediction: RacePrediction,
    conditions: CourseConditions,
) -> RacePrediction:
    """
    Apply all condition adjustments to a base prediction.
    Returns a new RacePrediction with adjustments applied.
    """
    adjustments: dict[str, float] = {}
    distance = prediction.distance_m

    # Calculate each adjustment
    temp_adj = _temperature_adjustment(conditions.temperature_f, distance)
    if abs(temp_adj) > 0.01:
        adjustments["temperature"] = round(temp_adj, 2)

    humid_adj = _humidity_adjustment(conditions.humidity_pct, conditions.temperature_f)
    if abs(humid_adj) > 0.01:
        adjustments["humidity"] = round(humid_adj, 2)

    wind_adj = _wind_adjustment(conditions.wind_mph, conditions.wind_direction)
    if abs(wind_adj) > 0.01:
        adjustments["wind"] = round(wind_adj, 2)

    elev_adj = _elevation_adjustment(
        conditions.elevation_gain_m,
        conditions.elevation_loss_m,
        distance,
    )
    if abs(elev_adj) > 0.01:
        adjustments["elevation"] = round(elev_adj, 2)

    surface_adj = _surface_adjustment(conditions.surface)
    if abs(surface_adj) > 0.01:
        adjustments["surface"] = round(surface_adj, 2)

    shoe_adj = _shoe_adjustment(conditions.shoe_type, conditions.surface, distance)
    if abs(shoe_adj) > 0.01:
        adjustments["shoes"] = round(shoe_adj, 2)

    alt_adj = _altitude_adjustment(conditions.altitude_m, distance)
    if abs(alt_adj) > 0.01:
        adjustments["altitude"] = round(alt_adj, 2)

    ready_adj = _readiness_adjustment(
        conditions.hrv_status,
        conditions.sleep_score,
        conditions.body_battery,
    )
    if abs(ready_adj) > 0.01:
        adjustments["readiness"] = round(ready_adj, 2)

    # Total percentage adjustment
    total_pct = sum(adjustments.values())

    # Apply to times
    factor = 1 + total_pct / 100
    new_time = prediction.time_s * factor
    new_fast = prediction.fast_s * factor
    new_slow = prediction.slow_s * factor
    new_pace = new_time / (distance / 1609.34)

    return RacePrediction(
        distance_m=distance,
        distance_label=prediction.distance_label,
        time_s=new_time,
        fast_s=new_fast,
        slow_s=new_slow,
        pace_per_mile_s=new_pace,
        model_contributions=prediction.model_contributions,
        adjustments_applied=adjustments,
        adjustment_pct=total_pct,
        theoretical_time_s=prediction.theoretical_time_s,
        volume_penalty_pct=prediction.volume_penalty_pct,
        readiness_status=prediction.readiness_status,
        readiness_note=prediction.readiness_note,
    )

DAYS_OF_WEEK = ["Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday"]


def _parse_race_day(day_str: str) -> str:
    s = day_str.strip().lower()
    for d in DAYS_OF_WEEK:
        if d.lower() in s:
            return d
    try:
        from datetime import datetime
        dt = datetime.fromisoformat(day_str.strip())
        return dt.strftime("%A")
    except Exception:
        return "Saturday"


def _dist_to_miles(dist_label: str) -> float:
    mapping = {
        "800m": 2.5,      # race + warm-up/cooldown
        "1600m": 3.0,
        "Mile": 3.0,
        "3200m": 4.0,
        "5K": 5.1,        # 3.1 mi race + 2.0 mi WU/CD
        "10K": 7.7,
        "Half Marathon": 14.1,
    }
    return mapping.get(dist_label, 4.5)

class GarminSync:
    """Manages connection and sync with Garmin Connect."""

    def __init__(self):
        self.client: Optional[Garmin] = None
        self._connected = False
        self.last_error: Optional[str] = None

    @property
    def is_connected(self) -> bool:
        return self._connected

    def connect(self) -> bool:
        """Connect to Garmin, preferring saved tokens to avoid rate limits."""
        token_dir = config.GARMIN_TOKEN_DIR

        # Try 1: Resume with saved tokens
        if os.path.exists(token_dir):
            try:
                client = Garmin()
                client.login(tokenstore=token_dir)
                self.client = client
                self._connected = True
                self.last_error = None
                try:
                    name = client.get_full_name()
                    log.info(f"Connected to Garmin via saved tokens ({name})")
                except Exception:
                    log.info("Connected to Garmin via saved tokens")
                return True
            except Exception as e:
                log.warning(f"Token login failed: {e}")

        # Try 2: Full login with credentials
        email = config.GARMIN_EMAIL
        password = config.GARMIN_PASSWORD
        if not email or not password:
            self.last_error = "No Garmin credentials or tokens found"
            log.warning(self.last_error)
            return False

        try:
            client = Garmin(email, password)
            client.login()
            try:
                os.makedirs(token_dir, exist_ok=True)
                client.garth.dump(token_dir)
                log.info(f"Saved auth tokens to {token_dir}")
            except Exception as e:
                log.warning(f"Could not persist tokens: {e}")
            self.client = client
            self._connected = True
            self.last_error = None
            log.info("Connected to Garmin via credentials")
            return True
        except Exception as e:
            self.last_error = f"Login failed: {e}"
            log.error(self.last_error)
            self._connected = False
            return False

    def _ensure_connected(self):
        if not self._connected or self.client is None:
            if not self.connect():
                raise ConnectionError(self.last_error or "Not connected to Garmin Connect")

    # ── Personal Records ─────────────────────────────────────────

    def sync_prs(self) -> int:
        """Pull personal records from Garmin Connect."""
        self._ensure_connected()
        count = 0

        try:
            pr_data = self.client.get_personal_record()
            if not pr_data:
                log.info("No PR data returned from Garmin")
                return 0

            records = pr_data if isinstance(pr_data, list) else pr_data.get("personal_records", [pr_data])

            for record in records:
                type_id = record.get("typeId") or record.get("type_id")
                distance = None
                label = ""

                # 1. Match by numeric typeId
                if type_id and type_id in GARMIN_TYPE_ID_MAP:
                    distance, label = GARMIN_TYPE_ID_MAP[type_id]

                # 2. Match by explicit distance / name if available
                if not distance:
                    pr_name = str(record.get("pr_name") or record.get("typeKey") or record.get("personalRecordType") or "").lower()
                    dist_field = record.get("distance_m") or record.get("distance")
                    if dist_field:
                        distance = float(dist_field)
                        label = str(record.get("pr_name") or record.get("label") or f"{int(distance)}m")
                    elif "1k" in pr_name or "1_km" in pr_name:
                        distance, label = 1000.0, "1K"
                    elif "mile" in pr_name and "5" not in pr_name:
                        distance, label = 1609.344, "Mile"
                    elif "5k" in pr_name or "5_km" in pr_name:
                        distance, label = 5000.0, "5K"
                    elif "10k" in pr_name or "10_km" in pr_name:
                        distance, label = 10000.0, "10K"
                    elif "half" in pr_name:
                        distance, label = 21097.5, "Half Marathon"
                    elif "marathon" in pr_name:
                        distance, label = 42195.0, "Marathon"

                pr_val = record.get("value") or record.get("value_raw") or record.get("prValue")
                if distance and pr_val and float(pr_val) > 0:
                    raw_date = (
                        record.get("actStartDateTimeInGMTFormatted")
                        or record.get("prStartTimeGmtFormatted")
                        or record.get("date_gmt")
                        or record.get("prDate")
                        or ""
                    )
                    date_str = str(raw_date)[:10]

                    effort = {
                        "distance_m": distance,
                        "time_s": float(pr_val),
                        "date": date_str,
                        "activity_id": None,
                        "garmin_id": record.get("activityId") or record.get("activity_id"),
                        "power_w": None,
                        "hr_avg": None,
                        "source": "garmin_pr",
                        "label": label,
                    }
                    db.upsert_effort(effort)
                    count += 1

            db.log_sync("prs", "success", f"Synced {count} PRs")
            log.info(f"Synced {count} PRs from Garmin")
        except Exception as e:
            db.log_sync("prs", "error", str(e))
            log.error(f"PR sync failed: {e}")

        return count

    # ── Health data ──────────────────────────────────────────────

    def sync_health(self, date_str: str | None = None) -> bool:
        """Pull physiological and recovery metrics from Garmin Connect."""
        self._ensure_connected()
        target_date = date_str or datetime.now().strftime("%Y-%m-%d")

        health_data: dict = {
            "date": target_date,
            "vo2max": None, "hrv_last_night": None, "hrv_weekly_avg": None,
            "hrv_status": None, "hrv_baseline_low": None, "hrv_baseline_high": None,
            "sleep_score": None, "sleep_seconds": None, "deep_sleep_s": None,
            "rem_sleep_s": None, "body_battery": None, "body_battery_high": None,
            "body_battery_low": None, "readiness_score": None, "readiness_level": None,
            "training_status": None, "training_status_phrase": None,
            "load_anaerobic": None, "load_aerobic_low": None, "load_aerobic_high": None,
            "resting_hr": None, "stress_avg": None, "raw_json": "{}",
        }

        raw = {}

        try:
            # 1. HRV
            try:
                hrv = self.client.get_hrv_data(target_date)
                raw["hrv"] = hrv
                if hrv:
                    summary = hrv.get("hrvSummary", {})
                    health_data["hrv_last_night"] = summary.get("lastNightAvg")
                    health_data["hrv_weekly_avg"] = summary.get("weeklyAvg")
                    health_data["hrv_status"] = summary.get("status")
                    baseline = summary.get("baseline", {})
                    health_data["hrv_baseline_low"] = baseline.get("balancedLow")
                    health_data["hrv_baseline_high"] = baseline.get("balancedUpper")
            except Exception as e:
                log.warning(f"HRV fetch failed: {e}")

            # 2. Sleep
            try:
                sleep = self.client.get_sleep_data(target_date)
                raw["sleep"] = sleep
                if sleep:
                    daily = sleep.get("dailySleepDTO", sleep)
                    scores = daily.get("sleepScores", {})
                    overall = scores.get("overall", {})
                    health_data["sleep_score"] = overall.get("value")
                    health_data["sleep_seconds"] = daily.get("sleepTimeSeconds")
                    health_data["deep_sleep_s"] = daily.get("deepSleepSeconds")
                    health_data["rem_sleep_s"] = daily.get("remSleepSeconds")
            except Exception as e:
                log.warning(f"Sleep fetch failed: {e}")

            # 3. Body Battery
            try:
                bb = self.client.get_body_battery(target_date)
                raw["body_battery"] = bb
                if bb and isinstance(bb, list) and bb:
                    item = bb[0]
                    health_data["body_battery_high"] = item.get("charged")
                    health_data["body_battery_low"] = item.get("drained")
                    vals = item.get("bodyBatteryValuesArray", [])
                    if vals:
                        last_pt = vals[-1]
                        health_data["body_battery"] = last_pt[-1] if isinstance(last_pt, list) else last_pt
            except Exception as e:
                log.warning(f"Body battery fetch failed: {e}")

            # 4. Stats (Resting HR & Stress)
            try:
                stats = self.client.get_stats(target_date)
                raw["stats"] = stats
                if stats:
                    health_data["resting_hr"] = stats.get("restingHeartRate")
                    health_data["stress_avg"] = stats.get("averageStressLevel")
            except Exception as e:
                log.warning(f"Stats fetch failed: {e}")

            # 5. VO2 Max
            try:
                vo2 = self.client.get_max_metrics(target_date)
                raw["vo2max"] = vo2
                if vo2 and isinstance(vo2, list):
                    for entry in vo2:
                        generic = entry.get("generic", {})
                        if generic:
                            health_data["vo2max"] = generic.get("vo2MaxPreciseValue") or generic.get("vo2MaxValue")
                            break
            except Exception as e:
                log.warning(f"VO2max fetch failed: {e}")

            # 6. Training Status & Load Balance
            try:
                ts = self.client.get_training_status(target_date)
                raw["training_status"] = ts
                if ts:
                    latest = ts.get("mostRecentTrainingStatus", {})
                    status_data = latest.get("latestTrainingStatusData", {})
                    for device_data in status_data.values():
                        health_data["training_status"] = device_data.get("trainingStatus")
                        health_data["training_status_phrase"] = device_data.get("trainingStatusFeedbackPhrase")

                    balance = ts.get("mostRecentTrainingLoadBalance", {})
                    for balance_map in balance.get("metricsTrainingLoadBalanceDTOMap", {}).values():
                        health_data["load_anaerobic"] = balance_map.get("monthlyLoadAnaerobic")
                        health_data["load_aerobic_low"] = balance_map.get("monthlyLoadAerobicLow")
                        health_data["load_aerobic_high"] = balance_map.get("monthlyLoadAerobicHigh")
            except Exception as e:
                log.warning(f"Training status fetch failed: {e}")

            health_data["raw_json"] = json.dumps(raw, default=str)
            db.upsert_health(health_data)
            db.log_sync("health", "success", f"Health data for {target_date}")
            log.info(f"Synced health data for {target_date}")
            return True

        except Exception as e:
            db.log_sync("health", "error", str(e))
            log.error(f"Health sync failed: {e}")
            return False

        try:
            results["prs"] = self.sync_prs()
        except Exception as e:
            results["errors"].append(f"PRs: {e}")

        try:
            results["health"] = self.sync_health()
        except Exception as e:
            results["errors"].append(f"Health: {e}")

        try:
            yesterday = (datetime.now() - timedelta(days=1)).strftime("%Y-%m-%d")
            self.sync_health(yesterday)
        except Exception:
            pass

        return results

def chat_with_coach(
    message: str,
    history: list[dict[str, str]],
    context: dict[str, Any],
) -> dict[str, Any]:
    """
    Main chat handler. Attempts Gemini API if GEMINI_API_KEY is configured,
    otherwise uses the built-in physiological coaching engine.
    """
    key = os.getenv("GEMINI_API_KEY", "") or getattr(config, "GEMINI_API_KEY", "")
    system_prompt = _build_system_prompt(context)

    # Attempt Gemini API call if key is present
    if key and len(key) > 10:
        try:
            url = f"https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key={key}"
            
            contents = []
            for h in history[-8:]:
                role = "user" if h.get("role") == "user" else "model"
                contents.append({"role": role, "parts": [{"text": h.get("content", "")}]})
            contents.append({"role": "user", "parts": [{"text": message}]})

            payload = {
                "system_instruction": {"parts": [{"text": system_prompt}]},
                "contents": contents,
                "tools": TOOLS_SCHEMA,
                "generationConfig": {
                    "temperature": 0.3,
                    "maxOutputTokens": 1024,
                },
            }

            resp = requests.post(url, json=payload, timeout=8)
            if resp.status_code == 200:
                data = resp.json()
                candidate = data.get("candidates", [{}])[0]
                content = candidate.get("content", {})
                parts = content.get("parts", [])

                reply_text = ""
                tool_call = None

                for p in parts:
                    if "text" in p:
                        reply_text += p["text"]
                    if "functionCall" in p:
                        fc = p["functionCall"]
                        tool_call = {
                            "name": fc.get("name"),
                            "parameters": fc.get("args", {}),
                        }

                # If Gemini returned a function call, execute it
                schedule_updated = False
                updated_plan = None
                if tool_call and tool_call.get("name") == "update_training_plan":
                    args = tool_call.get("parameters", {})
                    race_name = args.get("race_name", "Target Race")
                    race_dist = args.get("race_dist", "5K")
                    race_day = args.get("race_day", "Thursday")
                    active_focus = context.get("focus", "track_mid")
                    zones = context.get("zones")
                    base_plan = generate_training_plan(focus=active_focus, zones=zones)
                    easy_p = (zones or {}).get("easy", {}).get("label", "8:00–8:30/mi")
                    updated_plan = apply_race_taper(base_plan, race_name, race_dist, race_day, easy_pace=easy_p)
                    schedule_updated = True
                    if not reply_text:
                        reply_text = f"🎯 I have updated your training plan for **{race_name} ({race_dist}) on {race_day}**! The preceding 3–4 days have been auto-tapered with reduced volume, sharpening intervals, and pre-race shakeout strides."

                return {
                    "reply": reply_text,
                    "model": "gemini-2.5-flash",
                    "tool_calls": [tool_call] if tool_call else [],
                    "schedule_updated": schedule_updated,
                    "plan": updated_plan.to_dict() if updated_plan else None,
                }
        except Exception as e:
            log.warning(f"Gemini API request failed, falling back to physiological engine: {e}")

    # Built-in physiological reasoning fallback
    reply, tool_call = _generate_built_in_response(message, context)
    schedule_updated = False
    updated_plan = None

    if tool_call and tool_call.get("name") == "update_training_plan":
        args = tool_call.get("parameters", {})
        race_name = args.get("race_name", "Run for Pie 5K")
        race_dist = args.get("race_dist", "5K")
        race_day = args.get("race_day", "Thursday")
        active_focus = context.get("focus", "track_mid")
        zones = context.get("zones")
        base_plan = generate_training_plan(focus=active_focus, zones=zones)
        easy_p = (zones or {}).get("easy", {}).get("label", "8:00–8:30/mi")
        updated_plan = apply_race_taper(base_plan, race_name, race_dist, race_day, easy_pace=easy_p)
        schedule_updated = True

    return {
        "reply": reply,
        "model": "physiological-expert-v2",
        "tool_calls": [tool_call] if tool_call else [],
        "schedule_updated": schedule_updated,
        "plan": updated_plan.to_dict() if updated_plan else None,
    }

document.addEventListener("DOMContentLoaded", async () => {
    if (window.location.protocol === "file:") {
        document.getElementById("fileProtocolWarning").style.display = "block";
        return;
    }

    buildDistanceSelector();
    registerPWA();

    await Promise.all([
        loadOptions(),
        loadStatus(),

        loadFitness(),
        loadEfforts(),
        loadTrainingPlan(),
    ]);

    await refreshPredictions();
});

function registerPWA() {
    const isStandalone = window.matchMedia("(display-mode: standalone)").matches || window.navigator.standalone === true;
    if (isStandalone) {
        const btn = document.getElementById("installBtn");
        if (btn) btn.style.display = "none";
        return;
    }
    if ("serviceWorker" in navigator) {
        navigator.serviceWorker.register("/static/sw.js").catch(() => {});
    }
    window.addEventListener("beforeinstallprompt", (e) => {
        if (window.matchMedia("(display-mode: standalone)").matches || window.navigator.standalone === true) {
            return;
        }
        e.preventDefault();
        state.deferredPrompt = e;
        const btn = document.getElementById("installBtn");
        if (btn) btn.style.display = "flex";
    });
}

async function installPWA() {
    if (!state.deferredPrompt) return;
    state.deferredPrompt.prompt();
    await state.deferredPrompt.userChoice;
    state.deferredPrompt = null;
    document.getElementById("installBtn").style.display = "none";
}

function showToast(msg, type = "info") {
    const container = document.getElementById("toastContainer");
    const el = document.createElement("div");
    el.className = `toast ${type}`;
    el.textContent = msg;
    container.appendChild(el);
    setTimeout(() => {
        el.style.opacity = "0";
        el.style.transform = "translateY(-10px)";
        setTimeout(() => el.remove(), 300);
    }, 3600);
}

        .replace(/&/g, "&amp;")

        .replace(/"/g, "&quot;");
}

function updateSliderDisplay(input) {
    const val = input.value;
    if (input.id === "condTemp") document.getElementById("condTempVal").textContent = `${val}°F`;
    if (input.id === "condWind") document.getElementById("condWindVal").textContent = `${val} mph`;
    if (input.id === "condElevation") document.getElementById("condElevationVal").textContent = `${val} ft`;

        const tEl = document.getElementById("condTemp");
        const hEl = document.getElementById("condHumidity");

        showToast(`🌤️ Live conditions applied: ${data.temp_f}°F · ${data.humidity}% RH · ${data.wind_mph} mph wind`, "success");
        refreshPredictions();
    } catch (e) {
        showToast(e.message, "error");
    }
}

async function loadHeroPrediction(params) {
    try {
        const res = await fetch(`/api/predict?${params.toString()}`);
        const data = await res.json();
        const p = data.prediction;
        if (!p) return;
        state.currentPrediction = p;

        document.getElementById("heroTime").textContent = p.time_fmt;
        document.getElementById("heroFast").textContent = p.fast_fmt;
        document.getElementById("heroSlow").textContent = p.slow_fmt;
        document.getElementById("heroPace").innerHTML = `Goal Pace: <strong>${escapeHtml(p.pace_per_mile)} / mi</strong>`;

        const adjBadge = document.getElementById("heroAdj");
        adjBadge.style.display = "inline-block";
        if (p.adjustment_pct > 0.1) {
            adjBadge.className = "hero-adjustment penalty";
            adjBadge.textContent = `+${p.adjustment_pct.toFixed(1)}% slower from conditions`;
        } else if (p.adjustment_pct < -0.1) {
            adjBadge.className = "hero-adjustment benefit";
            adjBadge.textContent = `${p.adjustment_pct.toFixed(1)}% faster from conditions`;
        } else {
            adjBadge.className = "hero-adjustment neutral";
            adjBadge.textContent = "Neutral conditions (0.0%)";
        }

        const bonkEl = document.getElementById("bonkBanner");
        if (p.readiness_status !== "READY" && p.readiness_note) {
            bonkEl.style.display = "block";
            bonkEl.innerHTML = `<strong>⚠️ ${escapeHtml(p.readiness_status)}:</strong> ${escapeHtml(p.readiness_note)}`;
        } else {
            bonkEl.style.display = "none";
        }

        const adjWrap = document.getElementById("adjBreakdown");
        adjWrap.innerHTML = Object.entries(p.adjustments || {}).map(([k, v]) => {
            const cls = v > 0 ? "positive" : "negative";
            const sign = v > 0 ? "+" : "";
            return `<span class="adj-tag ${cls}">${escapeHtml(k)}: ${sign}${v}%</span>`;
        }).join("");

        renderModelBreakdown(p.models || {});
    } catch (e) {
        console.error("Hero prediction error", e);
    }
}

function renderModelBreakdown(models) {
    const box = document.getElementById("modelBreakdown");
    const labels = {
        calibrated_riegel: "Calibrated Riegel Curve",
        critical_speed: "Critical Speed + D' (Monod-Scherrer)",
        daniels_vdot: "Jack Daniels VDOT",
        standard_riegel: "Standard Riegel",
    };

    let html = `<div class="model-breakdown-title">Model Ensemble Breakdown (Pre-Condition Base)</div>`;
    for (const [k, v] of Object.entries(models)) {
        if (k === "_summary") continue;
        html += `
            <div class="model-row">
                <span class="model-name">${labels[k] || escapeHtml(k)} <span class="model-weight">(${Math.round((v.weight || 0) * 100)}%)</span></span>
                <span class="model-time">${escapeHtml(v.time_fmt)}</span>
            </div>
        `;
    }
    if (models._summary) {
        const s = models._summary;
        if (s.volume_penalty_pct > 0) {
            html += `
                <div class="model-row volume-penalty-row">
                    <span>Volume / Long-Run Durability Penalty</span>
                    <span>+${s.volume_penalty_pct}% (vs ${escapeHtml(s.theoretical_time_fmt)} theoretical)</span>
                </div>
            `;
        }
        html += `
            <div class="model-row base-summary-row">
                <span>Volume-Adjusted Base Prediction</span>
                <span class="model-time">${escapeHtml(s.base_time_fmt)}</span>
            </div>
        `;
    }
    box.innerHTML = html;
}

async function loadEfforts() {
    try {
        const res = await fetch("/api/efforts");
        const data = await res.json();
        const efforts = data.best_efforts || [];
        const grid = document.getElementById("effortsGrid");

        if (!efforts.length) {
            grid.innerHTML = `<div class="empty-state">No efforts recorded yet. Sync Garmin Connect to populate PRs.</div>`;
            return;
        }

        grid.innerHTML = efforts.map(e => {
            const t = e.time_s;
            const timeFmt = t >= 3600
                ? `${Math.floor(t / 3600)}:${String(Math.floor((t % 3600) / 60)).padStart(2, "0")}:${(t % 60).toFixed(1).padStart(4, "0")}`
                : `${Math.floor(t / 60)}:${(t % 60).toFixed(2).padStart(5, "0")}`;
            const paceS = t / (e.distance_m / 1609.344);
            const paceFmt = `${Math.floor(paceS / 60)}:${String(Math.round(paceS % 60)).padStart(2, "0")}/mi`;
            const lbl = e.label || `${Math.round(e.distance_m)}m`;

            return `
                <div class="effort-card">
                    <div class="left">
                        <div class="effort-dist">${escapeHtml(lbl)}</div>
                        <div class="effort-date">${escapeHtml(e.date || "Verified PR")}</div>
                    </div>
                    <div class="right">
                        <div class="effort-time">${timeFmt}</div>
                        <div class="effort-pace">${paceFmt}</div>
                    </div>
                </div>
            `;
        }).join("");
    } catch {}
}

async function loadStatus() {
    try {
        const res = await fetch("/api/status");
        const s = await res.json();
        const badge = document.getElementById("syncBadge");
        const txt = document.getElementById("syncText");

        if (s.syncing) {
            badge.className = "sync-badge";
            txt.textContent = "Syncing Garmin…";
        } else if (s.is_live) {
            badge.className = "sync-badge connected";
            txt.textContent = "Garmin Live";
        } else if (s.connected) {
            badge.className = "sync-badge connected";
            txt.textContent = "Garmin Cached";
        } else {
            badge.className = "sync-badge";
            txt.textContent = "Sync Garmin";
        }
        return s;
    } catch {
        return null;
    }
}

async function triggerSync() {
    const txt = document.getElementById("syncText");
    txt.textContent = "Syncing…";
    try {
        const res = await fetch("/api/sync", { method: "POST" });
        const data = await res.json();
        if (res.status === 409) {
            showToast(data.message, "info");
        } else {
            showToast("Syncing with Garmin Connect…", "info");
        }
        pollSyncCompletion();
    } catch (e) {
        showToast("Sync request failed", "error");
        loadStatus();
    }
}

function pollSyncCompletion() {
    clearInterval(state.syncPollTimer);
    state.syncPollTimer = setInterval(async () => {
        const s = await loadStatus();
        if (s && !s.syncing) {
            clearInterval(state.syncPollTimer);
            if (s.sync_status === "error") {
                showToast(s.sync_message || "Garmin sync failed", "error");
            } else {
                showToast(s.sync_message || "Garmin sync complete!", "success");
                await Promise.all([loadFitness(), loadMorningBriefing(), loadEfforts(), refreshPredictions()]);
            }
        }
    }, 1500);
}

function formatCoachReply(text) {
    return escapeHtml(text)
        .replace(/\*\*(.+?)\*\*/g, "<strong>$1</strong>")
        .replace(/`(.+?)`/g, "<code>$1</code>")
        .replace(/\n/g, "<br>");
}

async function submitCoachMessage(message) {
    const msgsEl = document.getElementById("coachMessages");
    msgsEl.innerHTML += `<div class="coach-msg user"><div class="msg-bubble">${escapeHtml(message)}</div></div>`;
    const loadingId = `coach-wait-${Date.now()}`;
    msgsEl.innerHTML += `<div class="coach-msg assistant" id="${loadingId}"><div class="msg-bubble">Analyzing telemetry…</div></div>`;
    msgsEl.scrollTop = msgsEl.scrollHeight;

    state.chatHistory.push({ role: "user", content: message });

    const context = {
        surface: document.getElementById("condSurface").value,
        surface: document.getElementById("condSurface").value,
        shoe: document.getElementById("condShoe").value,
        focus: document.getElementById("planFocusSelect").value,
        weather: {
            temp_f: Number(document.getElementById("condTemp").value),
            humidity: Number(document.getElementById("condHumidity").value),
            wind_mph: Number(document.getElementById("condWind").value),
            wind_dir: document.getElementById("condWindDir").value,
        },
        prediction: state.currentPrediction || {},
        training_plan: state.currentPlan || {},
    };

    try {
        const res = await fetch("/api/chat", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ message, history: state.chatHistory, context }),
        });
        const data = await res.json();
        const waitEl = document.getElementById(loadingId);
        if (waitEl) waitEl.remove();

        const reply = data.reply || "No response received.";
        state.chatHistory.push({ role: "assistant", content: reply });

        let extraBanner = "";
        if (data.schedule_updated && data.plan) {
            renderTrainingPlan(data.plan);
            extraBanner = `<div class="coach-tool-banner">📅 Training schedule automatically updated for race week!</div>`;
            showToast("🤖 Coach updated your training plan!", "success");
        }

        msgsEl.innerHTML += `
            <div class="coach-msg assistant">
                <div class="msg-bubble">${formatCoachReply(reply)}${extraBanner}</div>
            </div>
        `;
        msgsEl.scrollTop = msgsEl.scrollHeight;
    } catch (e) {
        const waitEl = document.getElementById(loadingId);
        if (waitEl) waitEl.innerHTML = `<div class="msg-bubble">Connection error reaching AI Coach.</div>`;
    }

// RaceKit Service Worker for PWA installation on Chromebook/Android/Desktop
const CACHE_NAME = 'racekit-v2';
const ASSETS = [
    '/',
    '/static/index.html',
    '/static/style.css',
    '/static/app.js',
    '/static/manifest.json',
    '/static/icon.svg',
    '/static/icon-192.png',
    '/static/icon-512.png'
];

self.addEventListener('install', (event) => {
    event.waitUntil(
        caches.open(CACHE_NAME).then((cache) => cache.addAll(ASSETS))
    );
    self.skipWaiting();
});

self.addEventListener('activate', (event) => {
    event.waitUntil(
        caches.keys().then((keys) => {
            return Promise.all(
                keys.filter((key) => key !== CACHE_NAME).map((key) => caches.delete(key))
            );
        })
    );
    self.clients.claim();
});

self.addEventListener('fetch', (event) => {
    // API calls go directly to network
    if (event.request.url.includes('/api/')) {
        event.respondWith(fetch(event.request));
        return;
    }
    // Static assets: cache with network fallback
    event.respondWith(
        caches.match(event.request).then((cached) => {
            return cached || fetch(event.request).catch(() => caches.match('/'));
        })
    );
});

#!/usr/bin/env bash
# RaceKit — Start script
set -e

cd "$(dirname "$0")"

# Check virtual environment
if [ ! -d ".venv" ]; then
    echo "🏗️  Creating virtual environment..."
    python3 -m venv .venv
fi

# Ensure data dir
mkdir -p data

# Check for stop mode
if [ "$1" == "--stop" ] || [ "$1" == "stop" ]; then
    if [ -f data/racekit.pid ]; then
        PID=$(cat data/racekit.pid)
        if kill -0 "$PID" 2>/dev/null; then
            kill "$PID"
            echo "🛑 RaceKit server (PID $PID) stopped."
            rm -f data/racekit.pid
            exit 0
        fi
    fi
    pkill -f "python.*app.py" 2>/dev/null && echo "🛑 RaceKit processes stopped." || echo "RaceKit is not running."
    exit 0
fi

# Check for daemon mode
if [ "$1" == "--daemon" ] || [ "$1" == "-d" ]; then
    if curl -s --max-time 1 "http://127.0.0.1:8420/api/status" >/dev/null 2>&1; then
        echo "🏃 RaceKit is already running on http://127.0.0.1:8420"
        exit 0
    fi
    echo "🏃 Starting RaceKit daemon on http://0.0.0.0:8420 (logs: data/racekit.log)..."
    nohup .venv/bin/python app.py > data/racekit.log 2>&1 &
    PID=$!
    echo $PID > data/racekit.pid
    # wait for startup
    for i in {1..20}; do
        if curl -s --max-time 1 "http://127.0.0.1:8420/api/status" >/dev/null 2>&1; then
            echo "✅ RaceKit is live! (PID $PID)"
            exit 0
        fi
        sleep 0.2
    done
    echo "⚠️ Started, check data/racekit.log for status."
    exit 0
fi

echo ""
echo "=========================================================="
echo "🏃  RaceKit — Precision Race Predictor & Live Garmin Engine"
echo "    http://127.0.0.1:8420"
echo "=========================================================="
echo "    Press Ctrl+C to stop"
echo ""

# Start server
exec .venv/bin/python app.py

# 🏃 RaceKit — Precision Race Predictor & Garmin Training Engine

RaceKit is a high-performance running performance predictor, Garmin Connect sync engine, and verified training plan builder with race-week auto-tapering and an integrated AI Coach.

---

## ⚡ Quick Start

### 1. Requirements
- Python 3.10+
- Linux, macOS, or Windows (WSL / PowerShell)

### 2. Launching RaceKit
Run the provided start script:
```bash
./start.sh
```

Or manually:
```bash
python3 -m venv .venv
source .venv/bin/activate    # On Windows: .venv\Scripts\activate
pip install -r requirements.txt
python3 app.py
```

Open your browser to:
**[http://localhost:8420](http://localhost:8420)**

You can also run it in the background as a daemon:
```bash
./start.sh --daemon
# To stop:
./start.sh --stop
```

---

## 🚀 Key Features

1. **Multi-Model Race Prediction Engine:**
   - Calibrated Riegel power law with distance decay.
   - 2-Parameter Critical Speed model ($D' / S_c$).
   - Jack Daniels VDOT formula.
   - Weighted ensemble predictions across 800m, 1600m, 1 Mile, 3200m, 5K, 10K, Half Marathon, and Marathon.

2. **Garmin Connect Live Sync:**
   - Real-time synchronization of activities, personal records, VO2max, HRV status, sleep scores, and Body Battery.
   - Works online and offline with local SQLite database caching.

3. **5-Shoe Rotation & Surface Dynamics:**
   - **Nike Dragonfly XC** — XC / Grass Spikes with ZoomX energy return.
   - **Nike Vaporfly Next% 2** — Carbon road racing super-shoe.
   - **Nike Dragonfly 2** — Track spikes for 800m–5000m.
   - **Nike Streakfly** — Lightweight workout flat / PR anchor shoe.
   - **New Balance 1080 v15** — Max-cushion daily trainer baseline.
   - Surface auto-matching: Selecting Road, Track, or XC automatically equips the optimal race shoe.

4. **Verified Training Plans with Auto-Taper:**
   - Training plans tailored for Track Mid-Distance, Track Long-Distance, Cross Country (38–42 mi/wk), 10K, and Half Marathon.
   - Target Race input bar: Set your race date, name, and distance (e.g. *Run for Pie 5K — Thursday*).
   - Automatically re-architects the week with reduced volume, light shakeout, strides, and rest.

5. **Integrated AI Coach Drawer:**
   - Real-time advice on race strategy, recovery analysis, and shoe choice.
   - One-click quick chips for pacing, readiness, and taper planning.

6. **Installable Desktop PWA:**
   - Can be installed as a standalone desktop app on Chrome OS (Chromebook), Linux, Windows, or macOS.
